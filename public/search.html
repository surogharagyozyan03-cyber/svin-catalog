document.addEventListener('DOMContentLoaded', () => { const catBtn = document.getElementById('categoriesBtn'); const catMenu = document.getElementById('categoriesMenu'); const dropdownWrapper = catBtn ? catBtn.closest('.dropdown') : null; if (catBtn && catMenu && dropdownWrapper) { catBtn.addEventListener('click', (e) => { e.stopPropagation(); catMenu.classList.toggle('show'); dropdownWrapper.classList.toggle('active'); }); document.addEventListener('click', (e) => { if (!dropdownWrapper.contains(e.target)) { catMenu.classList.remove('show'); dropdownWrapper.classList.remove('active'); } }); } const searchInput = document.getElementById('searchInput'); const searchBtn = document.getElementById('searchBtn'); const searchTitle = document.getElementById('searchTitle'); const searchSubtitle = document.getElementById('searchSubtitle'); const grid = document.getElementById('searchResultsGrid'); // Функция поиска и отрисовки результатов const runSearch = (queryText) => { const query = (queryText || '').trim(); if (!grid) return; // Если мы не на странице с сеткой результатов if (!query) { if (searchTitle) searchTitle.textContent = `Որոնման արդյունքներ`; if (searchSubtitle) searchSubtitle.textContent = `Մուտքագրեք որոնման բառը վերևի դաշտում`; grid.innerHTML = `
Խնդրում ենք որոնման դաշտում գրել բառ և սեղմել «Որոնել»:
`; return; } if (searchTitle) searchTitle.textContent = `Որոնում: "${query}"`; fetch('/api/products') .then(res => res.json()) .then(products => { const lowerQuery = query.toLowerCase(); const matchedProducts = products.filter(p => { const name = (p.name || '').toLowerCase(); const desc = (p.description || '').toLowerCase(); const cat = (p.category || '').toLowerCase(); const subcat = (p.subcategory || '').toLowerCase(); return name.includes(lowerQuery) || desc.includes(lowerQuery) || cat.includes(lowerQuery) || subcat.includes(lowerQuery); }); if (matchedProducts.length > 0) { if (searchSubtitle) searchSubtitle.textContent = `Գտնվել է ապրանք: ${matchedProducts.length}`; grid.innerHTML = ''; matchedProducts.forEach(prod => { const card = document.createElement('div'); card.className = 'product-card'; card.addEventListener('click', () => { if (typeof openModal === 'function') { openModal(prod); } }); card.innerHTML = `
${prod.subcategory || prod.category || 'Ապրանք'}
${prod.name || ''}
${prod.price ? Number(prod.price).toLocaleString() : '0'} ֏
`; grid.appendChild(card); }); } else { if (searchSubtitle) searchSubtitle.textContent = `Արդյունքներ չեն գտնվել`; grid.innerHTML = `
Ձեր որոնմանը համապատասխան ապրանքներ չեն գտնվել:
`; } }) .catch(err => { console.error("Search error:", err); if (searchSubtitle) searchSubtitle.textContent = `Սխալ տվյալները ստանալիս`; }); }; // 1. Проверяем URL при загрузке страницы (?q=...) const urlParams = new URLSearchParams(window.location.search); const initialQuery = urlParams.get('q') || ''; if (searchInput) { searchInput.value = initialQuery; } // Запускаем поиск по переданному параметру из URL runSearch(initialQuery); // 2. Слушаем события ввода if (searchBtn && searchInput) { const handleSearchAction = () => { const val = searchInput.value.trim(); // Если мы уже на странице search.html — ищем на месте без перезагрузки if (window.location.pathname.includes('search.html')) { // Обновляем URL без перезагрузки страницы const newUrl = window.location.protocol + "//" + window.location.host + window.location.pathname + '?q=' + encodeURIComponent(val); window.history.pushState({ path: newUrl }, '', newUrl); runSearch(val); } else { // Если мы на главной или другой странице — переходим на search.html window.location.href = `search.html?q=${encodeURIComponent(val)}`; } }; searchBtn.addEventListener('click', handleSearchAction); searchInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') { handleSearchAction(); } }); } });
