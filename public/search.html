// Глобальная функция открытия модального окна (чтобы вызывалась отовсюду) function openModal(prod) { const modal = document.getElementById('productModal'); if (!modal) return; const modalImg = document.getElementById('modalImage'); const modalTitle = document.getElementById('modalTitle'); const modalCat = document.getElementById('modalCategory'); const modalDesc = document.getElementById('modalDescription'); const modalPrice = document.getElementById('modalPrice'); if (modalImg) modalImg.src = prod.image || 'logo.png'; if (modalTitle) modalTitle.textContent = prod.name || ''; if (modalCat) modalCat.textContent = prod.subcategory || prod.category || 'Ապրանք'; if (modalDesc) modalDesc.textContent = prod.description || ''; if (modalPrice) modalPrice.textContent = `${prod.price ? Number(prod.price).toLocaleString() : '0'} ֏`; modal.classList.add('active'); // Или modal.style.display = 'flex'; в зависимости от твоих CSS-стилей modal.style.display = 'flex'; } // Функция закрытия модального окна function closeModal() { const modal = document.getElementById('productModal'); if (modal) { modal.classList.remove('active'); modal.style.display = 'none'; } } document.addEventListener('DOMContentLoaded', () => { // --- 1. Выпадающее меню категорий --- const catBtn = document.getElementById('categoriesBtn'); const catMenu = document.getElementById('categoriesMenu'); const dropdownWrapper = catBtn ? catBtn.closest('.dropdown') : null; if (catBtn && catMenu && dropdownWrapper) { catBtn.addEventListener('click', (e) => { e.stopPropagation(); catMenu.classList.toggle('show'); dropdownWrapper.classList.toggle('active'); }); document.addEventListener('click', (e) => { if (!dropdownWrapper.contains(e.target)) { catMenu.classList.remove('show'); dropdownWrapper.classList.remove('active'); } }); } // --- 2. Обработчики закрытия модального окна --- const modal = document.getElementById('productModal'); const closeBtn = document.querySelector('.close-modal'); if (closeBtn) { closeBtn.addEventListener('click', closeModal); } if (modal) { window.addEventListener('click', (e) => { if (e.target === modal) { closeModal(); } }); } document.addEventListener('keydown', (e) => { if (e.key === 'Escape') { closeModal(); } }); // --- 3. Поиск и отрисовка карточек --- const searchInput = document.getElementById('searchInput'); const searchBtn = document.getElementById('searchBtn'); const searchTitle = document.getElementById('searchTitle'); const searchSubtitle = document.getElementById('searchSubtitle'); const grid = document.getElementById('searchResultsGrid'); const runSearch = (queryText) => { const query = (queryText || '').trim(); if (!grid) return; // Если не на странице результатов if (!query) { if (searchTitle) searchTitle.textContent = `Որոնման արդյունքներ`; if (searchSubtitle) searchSubtitle.textContent = `Մուտքագրեք որոնման բառը վերևի դաշտում`; grid.innerHTML = `
Խնդրում ենք որոնման դաշտում գրել բառ և սեղմել «Որոնել»:
`; return; } if (searchTitle) searchTitle.textContent = `Որոնում: "${query}"`; fetch('/api/products') .then(res => res.json()) .then(products => { const lowerQuery = query.toLowerCase(); const matchedProducts = products.filter(p => { const name = (p.name || '').toLowerCase(); const desc = (p.description || '').toLowerCase(); const cat = (p.category || '').toLowerCase(); const subcat = (p.subcategory || '').toLowerCase(); return name.includes(lowerQuery) || desc.includes(lowerQuery) || cat.includes(lowerQuery) || subcat.includes(lowerQuery); }); if (matchedProducts.length > 0) { if (searchSubtitle) searchSubtitle.textContent = `Գտնվել է ապրանք: ${matchedProducts.length}`; grid.innerHTML = ''; matchedProducts.forEach(prod => { const card = document.createElement('div'); card.className = 'product-card'; // Клик по карточке открывает модальное окно card.addEventListener('click', () => { openModal(prod); }); card.innerHTML = `
${prod.subcategory || prod.category || 'Ապրանք'}
${prod.name || ''}
${prod.price ? Number(prod.price).toLocaleString() : '0'} ֏
`; grid.appendChild(card); }); } else { if (searchSubtitle) searchSubtitle.textContent = `Արդյունքներ չեն գտնվել`; grid.innerHTML = `
Ձեր որոնմանը համապատասխան ապրանքներ չեն գտնվել:
`; } }) .catch(err => { console.error("Search error:", err); if (searchSubtitle) searchSubtitle.textContent = `Սխալ տվյալները ստանալիս`; }); }; // --- 4. Старт поиска из параметров URL --- const urlParams = new URLSearchParams(window.location.search); const initialQuery = urlParams.get('q') || ''; if (searchInput) { searchInput.value = initialQuery; } runSearch(initialQuery); // --- 5. Обработка ввода в инпут поиска --- if (searchBtn && searchInput) { const handleSearchAction = () => { const val = searchInput.value.trim(); if (window.location.pathname.includes('search.html')) { const newUrl = window.location.protocol + "//" + window.location.host + window.location.pathname + '?q=' + encodeURIComponent(val); window.history.pushState({ path: newUrl }, '', newUrl); runSearch(val); } else { window.location.href = `search.html?q=${encodeURIComponent(val)}`; } }; searchBtn.addEventListener('click', handleSearchAction); searchInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') { handleSearchAction(); } }); } });
